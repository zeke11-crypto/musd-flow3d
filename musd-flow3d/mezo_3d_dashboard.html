                        1. Cloudflare Worker / Vercel API (`api/verify-musd.js`)
                        
                    

                    
export default {
  async fetch(request) {
    if (request.method !== "POST") return new Response("Method Not Allowed", { status: 405 });
    const { txHash, invoiceId, expectedMerchant, expectedAmount } = await request.json();
    const MUSD = "0x118917a40faf1cd7a13db0ef56c86de7973ac503";
    const TRANSFER_TOPIC = "0xddf252ad1be2c89b69c2b068fc378daa952ba7f163c4a11628f55a4df523b3ef";

    // Query Mezo Testnet RPC (Chain ID 31611)
    const rpcRes = await fetch("https://rpc.test.mezo.org", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ jsonrpc: "2.0", id: 1, method: "eth_getTransactionReceipt", params: [txHash] })
    });
    const { result: receipt } = await rpcRes.json();
    if (!receipt || receipt.status !== "0x1") {
      return Response.json({ verified: false, error: "Transaction not confirmed on Mezo 31611" }, { status: 400 });
    }

    const musdLog = receipt.logs.find(
      l => l.address.toLowerCase() === MUSD && l.topics[0] === TRANSFER_TOPIC
    );
    if (!musdLog) {
      return Response.json({ verified: false, error: "No MUSD ERC-20 Transfer log found" }, { status: 400 });
    }

    const toAddress = "0x" + musdLog.topics[2].slice(26).toLowerCase();
    const rawAmount = BigInt(musdLog.data);
    return Response.json({
      verified: true,
      chainId: 31611,
      invoiceId,
      txHash,
      recipient: toAddress,
      amountWei: rawAmount.toString(),
      explorer: `https://explorer.test.mezo.org/tx/${txHash}`
    });
  }
};
